cmake_minimum_required(VERSION 3.22)
project(kid-testharness C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
find_package(KWayland REQUIRED)
find_package(X11 REQUIRED)
add_executable(kid-input kidinput.cpp)
target_link_libraries(kid-input Qt6::Core Qt6::Gui Plasma::KWaylandClient)
add_executable(kid-selfmove kidselfmove.cpp)
target_link_libraries(kid-selfmove X11::X11)
install(TARGETS kid-input kid-selfmove DESTINATION bin)
